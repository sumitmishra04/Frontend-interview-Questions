document.querySelectorAll('.accept-btn').forEach(button => {
  button.addEventListener('click', (event) => {
    const listItem = event.target.closest('li');
    const friendName = listItem.textContent.trim().split(' ')[0]

    addFriend(friendName);
    listItem.remove();
  });
});

function addFriend(name) {
  const friendsList = document.getElementById('friends-list-container');
  const friendItem = document.createElement('li');
  friendItem.textContent = name;

  const dateSpan = document.createElement('span');
  dateSpan.classList.add('friend-date');
  const date = new Date().getDate()
  const month = new Date().getMonth()
  const year = new Date().getFullYear()
  dateSpan.textContent = `(since ${year}-${month}-${date})`
  const removeButton = document.createElement('button');
  removeButton.textContent = 'Remove';
  removeButton.classList.add('remove-btn');
  removeButton.addEventListener('click', () => {
    friendItem.remove();
  });

  friendItem.appendChild(dateSpan);
  friendItem.appendChild(removeButton);
  friendsList.appendChild(friendItem);
}

const SAMPLE_RUN_ATTEMPTS = 10
const RANGE = 2

const measurePerf = async (fn, runAttempts = SAMPLE_RUN_ATTEMPTS, range=RANGE) => {
    const result = []
    
    if(runAttempts === null || runAttempts === undefined) {
        runAttempts = SAMPLE_RUN_ATTEMPTS
    }
    if(range === null || range === undefined) {
        range = RANGE
    }
    async function run() {
        const start = Date.now()
        await fn()
        const end = Date.now()
        result.push(end - start)
    }
    for(let i = 0; i< runAttempts;i++) {
        await run()
    }
    result.sort((a,b) => a-b)
    const threshold = result[Math.floor(runAttempts/2)] * range
    console.log('before',result)
    const filteredResult = result.filter(v => v<threshold)
    console.log('after',filteredResult)
    return {
        totalResult: result,
        filteredResult: filteredResult,
        value: filteredResult.reduce((acc, curr) => acc + curr, 0)/runAttempts  
    }  
}
