what are the memory leak points in promise.race

function promiseRace(promises) {
    return new Promise((resolve, reject) => {
        let settled = false;
        promises.forEach(p => {
            Promise.resolve(p)
                .then(value => {
                    if (!settled) {
                        settled = true;
                        resolve(value);
                    }
                })
                .catch(error => {
                    if (!settled) {
                        settled = true;
                        reject(error);
                    }
                });
        });
    });
}


| Issue                             | Cause                                                                            | Fix                                                             |
| --------------------------------- | -------------------------------------------------------------------------------- | --------------------------------------------------------------- |
| Memory leak / dangling closures   | All promises retain `.then(resolve)` and `.catch(reject)` even after one settles | Use a `settled` flag or an `AbortController`-style cancellation |
| Not critical for short-lived code | But accumulates in long-running systems                                          | Add guard logic to clean up references                          |
